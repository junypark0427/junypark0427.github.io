### 작성한 코드와 테스트

출금액이 잔액보다 크면 경고하고, 잔액에 음수를 대입하면 0으로 설정한다.

```python
class BankAccount:

    def __init__(self, owner):
        self.owner = owner
        self.__balance = 0

    def __str__(self):
        return f"{self.owner}님의 계좌 (잔액:{self.__balance})"

    @property
    def balance(self):
        return self.__balance

    @balance.setter
    def balance(self, balance):
        if balance < 0:
            self.__balance = 0
        else:
            self.__balance = balance

    def deposit(self, amount):
        if amount <= 0:
            print("입금액은 0보다 커야 합니다.")
        else:
            self.__balance += amount

    def withdraw(self, amount):
        if amount <= 0:
            print("출금액은 0보다 커야 합니다.")
            return
        if amount > self.__balance:
            print("잔액이 부족합니다.")
        else:
            self.__balance -= amount



A = BankAccount("A")
B = BankAccount("B")

A.balance = 100
print(A)
A.deposit(-10)
A.deposit(150)
print(A)

A.withdraw(300)
A.withdraw(100)
print(A)
```
